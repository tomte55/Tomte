```lua
-- The item link, count and kind of a loot message about yourself, or nil. patterns = { looted = Recap_LootPatterns
-- of the loot window and bonus roll strings, pushed = Recap_LootPatterns of the "You receive item" strings }.
function ns.Recap_LootMessage(text, patterns)
	for _, kind in ipairs(LOOT_KINDS) do
		local link, count = ns.Recap_LootLink(text, patterns[kind] or {})
		if link then
			return link, count, kind
		end
	end
	return nil
end

-- Whether a loot message is the session's loot. Loot windows and bonus rolls always are. Items pushed into the bags
-- (quest rewards and the like) are, unless a vendor, the mailbox, the auction house or a trade is open: then it's
-- a purchase or a delivery. Crafted items ("You create") never get here.
function ns.Recap_IsLoot(kind, ctx)
	if kind == "looted" then
		return true
	end
	return kind == "pushed" and not (ctx.vendor or ctx.mail or ctx.auction or ctx.trade)
end
```

```lua
-- After "Deposit for alts": how many of the tried stacks left the bags. tried = { { bag, slot, itemID, count } },
-- now(bag, slot) -> itemID, count, locked (what's in that slot now). An empty slot, another item or a smaller stack
-- moved; a locked slot is on its way (the server hasn't answered); the same item and count stayed.
-- -> moved, left, moving
function ns.Alts_DepositOutcome(tried, now)
	local moved, left, moving = 0, 0, 0
	for _, s in ipairs(tried) do
		local itemID, count, locked = now(s.bag, s.slot)
		if locked then
			moving = moving + 1
		elseif itemID ~= s.itemID or (count or 0) < s.count then
			moved = moved + 1
		else
			left = left + 1
		end
	end
	return moved, left, moving
end

local DEPOSIT_WHY = {
	closed = "the Warband bank can't be used right now",
	tabs = "you haven't bought a Warband bank tab",
	full = "the Warband bank is full",
}

local function Stacks(n)
	return ("%d stack%s"):format(n, n == 1 and "" or "s")
end

-- The chat line for "Deposit for alts": tried and moved count stacks; reason says why the rest stayed
-- ("closed" | "tabs" | "full" | nil when the game didn't say).
function ns.Alts_DepositText(tried, moved, reason)
	if tried == 0 then
		return "nothing to put in the Warband bank for your alts."
	end
	if moved >= tried then
		return ("put %s in the Warband bank for your alts."):format(Stacks(tried))
	end
	local why = DEPOSIT_WHY[reason] or "the game didn't move them"
	if moved == 0 then
		return ("couldn't put anything in the Warband bank: %s."):format(why)
	end
	return ("put %d of %s in the Warband bank for your alts; the rest stayed: %s."):format(moved, Stacks(tried), why)
end
```

```lua
-- "full" (no free slot; up front a full bank may still take stacks that merge into partial ones), else nil.
local function WarbandProblem(full)
	if not C_Bank.CanUseBank(Enum.BankType.Account) then
		return "closed"
	end
	if (C_Bank.FetchNumPurchasedBankTabs(Enum.BankType.Account) or 0) == 0 then
		return "tabs"
	end
	if full then
		local free = 0
		for _, bag in ipairs(WARBAND_TABS) do
			free = free + (C_Container.GetContainerNumFreeSlots(bag) or 0)
		end
		if free == 0 then
			return "full"
		end
	end
	return nil
end

local function SlotNow(bag, slot)
```
